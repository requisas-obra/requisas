/**
 * CREDENCIALES I2M – guarda el registro de la app de credenciales (la que está en GitHub)
 * en la hoja CRED_DATOS del registro. Sin cuenta de Claude: entra con PIN de Admin.
 * La página llama a este proyecto por POST (doPost).
 */
var CRED_HOJA = 'CRED_DATOS', CRED_TROZO = 45000;

function doPost(e) {
  var out;
  try {
    var b = JSON.parse(e.postData.contents || '{}');
    if (b.app === 'cred') out = credApi_(b);
    else throw new Error('Petición no válida.');
    out.ok = true;
  } catch (err) { out = { ok: false, error: String(err && err.message || err) }; }
  return ContentService.createTextOutput(JSON.stringify(out)).setMimeType(ContentService.MimeType.JSON);
}

function credApi_(b) {
  var u = usuario_(b.pin);
  if (!u.admin) throw new Error('Solo un Admin puede usar Credenciales.');
  var h = credHoja_();
  if (b.a === 'cargar') return credCargar_(h);
  var lock = LockService.getScriptLock(); lock.waitLock(30000);
  try {
    if (b.a === 'guardarPase') { credPoner_(h, 'pase', b.doc.id, b.doc); return {}; }
    if (b.a === 'borrarPase') { credBorrar_(h, 'pase', b.id); return {}; }
    if (b.a === 'guardarCfg') { credPoner_(h, 'config', 'general', b.cfg); return {}; }
    if (b.a === 'guardarLogo') { credPoner_(h, 'logo', b.k, { img: b.img }); return {}; }
    if (b.a === 'importar') { (b.pases || []).forEach(function (d) { credPoner_(h, 'pase', d.id, d); }); return { n: (b.pases || []).length }; }
  } finally { lock.releaseLock(); }
  throw new Error('Acción desconocida.');
}

function credHoja_() {
  var reg = registro_(), h = reg.getSheetByName(CRED_HOJA);
  if (!h) {
    h = reg.insertSheet(CRED_HOJA);
    h.getRange(1, 1, 1, 7).setValues([['Tipo', 'Id', 'N° control', 'Nombre', 'Puesto', 'Actualizado', 'Datos (no tocar)']])
      .setFontWeight('bold').setBackground('#FD665C').setFontColor('#FFFFFF');
    h.setFrozenRows(1);
  }
  return h;
}

function credFila_(h, tipo, id) {
  var n = h.getLastRow(); if (n < 2) return -1;
  var v = h.getRange(2, 1, n - 1, 2).getValues();
  for (var i = 0; i < v.length; i++) if (v[i][0] === tipo && String(v[i][1]) === String(id)) return i + 2;
  return -1;
}

function credPoner_(h, tipo, id, obj) {
  if (!id) throw new Error('Falta el id.');
  var s = JSON.stringify(obj), trozos = [];
  for (var i = 0; i < s.length; i += CRED_TROZO) trozos.push("'" + s.slice(i, i + CRED_TROZO));   // ' = que Sheets lo guarde como texto
  var fila = credFila_(h, tipo, id);
  if (fila === -1) fila = h.getLastRow() + 1;
  else { var w = h.getLastColumn(); if (w >= 7) h.getRange(fila, 7, 1, w - 6).clearContent(); }
  var cab = [tipo, String(id), obj.control || '', obj.nombre || '', obj.puesto || '', new Date()];
  if (h.getMaxColumns() < 6 + trozos.length) h.insertColumnsAfter(h.getMaxColumns(), 6 + trozos.length - h.getMaxColumns());
  h.getRange(fila, 1, 1, 6 + trozos.length).setValues([cab.concat(trozos)]);
}

function credBorrar_(h, tipo, id) { var f = credFila_(h, tipo, id); if (f !== -1) h.deleteRow(f); }

function credCargar_(h) {
  var out = { cfg: null, logos: {}, pases: [] }, n = h.getLastRow();
  if (n < 2) return out;
  var v = h.getRange(2, 1, n - 1, h.getLastColumn()).getValues();
  v.forEach(function (r) {
    var s = r.slice(6).join(''); if (!s) return;
    var o; try { o = JSON.parse(s); } catch (e) { return; }
    if (r[0] === 'pase') { o.id = String(r[1]); out.pases.push(o); }
    else if (r[0] === 'config') out.cfg = o;
    else if (r[0] === 'logo') out.logos[String(r[1])] = o.img || null;
  });
  return out;
}
